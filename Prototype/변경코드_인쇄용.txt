    public class Entry
    {
        public string Id = "";
        public DateTime Time;
        public string Title = "";
        public string Note = "";
        public bool Complete;
        public List<Item> Items = new List<Item>();
        public List<string> Photos = new List<string>();
        public override string ToString()
        {
            return Time.ToString("MM-dd HH:mm") + " " + Title + " "
                + Describe() + (Complete ? " [완료]" : " [진행]");
        }
        public string Describe()
        {
            List<string> parts = new List<string>();
            foreach (Item item in Items) parts.Add(item.ToString());
            return String.Join(", ", parts.ToArray());
        }
    }

        static readonly XmlSerializer Serializer = new XmlSerializer(typeof(Book));

        public static Book Clone(Book book)
        {
            using (MemoryStream memory = new MemoryStream())
            {
                WriteXml(memory, book);
                memory.Position = 0;
                return (Book)Serializer.Deserialize(memory);
            }
        }

        public static Book Load(string path)
        {
            if (!File.Exists(path)) return new Book();
            XmlReaderSettings options = new XmlReaderSettings();
            options.DtdProcessing = DtdProcessing.Prohibit;
            options.XmlResolver = null;
            using (XmlReader reader = XmlReader.Create(path, options))
            {
                Book book = (Book)Serializer.Deserialize(reader);
                Validate(book);
                return book;
            }
        }

        public static void Validate(Book book)
        {
            if (book == null || book.Version != 1 || book.Settings == null
                || book.Entries == null || book.PendingDays == null
                || book.AlarmAcknowledgements == null)
                throw new InvalidDataException("지원하지 않거나 손상된 원본입니다.");
            HashSet<string> ids = new HashSet<string>();
            foreach (Entry entry in book.Entries)
            {
                Guid parsedId;
                if (entry == null || !Guid.TryParseExact(entry.Id, "N", out parsedId)
                    || !ids.Add(entry.Id) || entry.Items == null || entry.Photos == null
                    || entry.Title == null || entry.Note == null || entry.Time.Year < 1753 || entry.Time.Year > 9998)
                    throw new InvalidDataException("기록 ID 또는 필드가 올바르지 않습니다.");
                HashSet<string> kinds = new HashSet<string>();
                foreach (Item item in entry.Items)
                    if (item == null || String.IsNullOrWhiteSpace(item.KindId)
                        || !kinds.Add(item.KindId) || item.Name == null || item.Unit == null
                        || item.Count < 0 || item.Count > 999)
                        throw new InvalidDataException("기록의 항목 또는 수량이 올바르지 않습니다.");
            }
            foreach (string day in book.PendingDays) ParseDay(day);
        }

        public static void AtomicWrite(string path, Action<Stream> write)
        {
            string temp = path + "." + Guid.NewGuid().ToString("N") + ".tmp";
            try
            {
                using (FileStream stream = new FileStream(temp,
                    FileMode.CreateNew, FileAccess.Write, FileShare.None))
                {
                    write(stream);
                    stream.Flush(true);
                }
                if (File.Exists(path)) File.Replace(temp, path, null);
                else File.Move(temp, path);
            }
            finally
            {
                try { if (File.Exists(temp)) File.Delete(temp); } catch { }
            }
        }

        public static void Save(string path, Book book)
        {
            Validate(book);
            AtomicWrite(path, delegate(Stream stream) { WriteXml(stream, book); });
        }

        static void WriteXml(Stream stream, Book book)
        {
            XmlWriterSettings settings = new XmlWriterSettings();
            settings.Encoding = new UTF8Encoding(false);
            settings.Indent = true;
            settings.NewLineHandling = NewLineHandling.Entitize;
            settings.CloseOutput = false;
            using (XmlWriter writer = XmlWriter.Create(stream, settings)) Serializer.Serialize(writer, book);
        }

        public static string Day(DateTime value) { return value.ToString("yyyy-MM-dd"); }
        public static DateTime ParseDay(string value)
        {
            return DateTime.ParseExact(value, "yyyy-MM-dd", CultureInfo.InvariantCulture);
        }
        public static void Queue(Book book, DateTime day)
        {
            string text = Day(day);
            if (!book.PendingDays.Contains(text)) book.PendingDays.Add(text);
        }

        public static string CsvCell(string value)
        {
            string text = value ?? "";
            string trimmed = text.TrimStart();
            if ((trimmed.Length > 0 && "=+-@".IndexOf(trimmed[0]) >= 0)
                || (text.Length > 0 && "\t\r\n".IndexOf(text[0]) >= 0))
                text = "'" + text;
            return "\"" + text.Replace("\"", "\"\"") + "\"";
        }

        static void CsvLine(StringBuilder text, List<string> fields)
        {
            for (int i = 0; i < fields.Count; i++)
            {
                if (i > 0) text.Append(',');
                text.Append(CsvCell(fields[i]));
            }
            text.Append("\r\n");
        }

        public static string ExportDay(Book book, DateTime day)
        {
            string folder = Path.Combine(book.Settings.CsvFolder, day.ToString("yyyy"), Day(day));
            Directory.CreateDirectory(folder);
            string path = Path.Combine(folder, "Records.csv");
            // IDs identify columns even after a type is renamed or removed.
            SortedDictionary<string, string> kinds = new SortedDictionary<string, string>();
            foreach (Kind kind in book.Settings.Kinds) kinds[kind.Id] = kind.Name;
            foreach (Entry entry in book.Entries)
                foreach (Item item in entry.Items)
                    if (!kinds.ContainsKey(item.KindId)) kinds[item.KindId] = item.Name;

            List<string> header = new List<string>(new string[] {
                "ID", "일시", "제목", "상태", "내역", "비고", "사진수" });
            foreach (KeyValuePair<string, string> kind in kinds)
                header.Add(kind.Value + " [" + kind.Key + "] 수량");
            StringBuilder text = new StringBuilder();
            CsvLine(text, header);
            foreach (Entry entry in book.Entries)
            {
                if (entry.Time.Date != day.Date) continue;
                List<string> row = new List<string>(new string[] {
                    entry.Id, entry.Time.ToString("yyyy-MM-dd HH:mm"), entry.Title,
                    entry.Complete ? "완료" : "진행", entry.Describe(), entry.Note,
                    entry.Photos.Count.ToString(CultureInfo.InvariantCulture) });
                foreach (string kind in kinds.Keys)
                {
                    int count = 0;
                    foreach (Item item in entry.Items)
                        if (item.KindId == kind) count += item.Count;
                    row.Add(count.ToString(CultureInfo.InvariantCulture));
                }
                CsvLine(text, row);
            }
            byte[] bytes = new UTF8Encoding(true).GetBytes(text.ToString());
            AtomicWrite(path, delegate(Stream stream)
            {
                byte[] bom = new UTF8Encoding(true).GetPreamble();
                stream.Write(bom, 0, bom.Length);
                stream.Write(bytes, 0, bytes.Length);
            });
            return path;
        }

        public static string SafeName(string value)
        {
            StringBuilder result = new StringBuilder();
            foreach (char c in value)
                result.Append(c < 32 || "<>:\"/\\|?*".IndexOf(c) >= 0 ? '_' : c);
            string name = result.ToString().Trim().TrimEnd('.', ' ');
            if (name.Length > 80) name = name.Substring(0, 80).TrimEnd('.', ' ');
            return name.Length == 0 ? "기록" : name;
        }

        public static string PhotoStem(Entry entry)
        {
            return SafeName(entry.Time.ToString("yyyyMMdd_HHmm") + " "
                + entry.Title + " " + entry.Describe()) + "_" + entry.Id.Substring(0, 8);
        }

        public static string PhotoFileName(Entry entry, int count, int ordinal, string extension)
        {
            if (count < 1 || ordinal < 1 || ordinal > count) throw new ArgumentOutOfRangeException("ordinal");
            return PhotoStem(entry) + (count > 1 ? " (" + ordinal.ToString(CultureInfo.InvariantCulture) + ")" : "") + extension;
        }

        public static string PhotoDestination(Entry entry, int count, int ordinal, string folder,
            string extension, string currentPath)
        {
            string path = Path.Combine(folder, PhotoFileName(entry, count, ordinal, extension));
            if (currentPath != null && String.Equals(Path.GetFullPath(path), Path.GetFullPath(currentPath),
                StringComparison.OrdinalIgnoreCase)) return path;
            if (File.Exists(path) || Directory.Exists(path))
                throw new IOException("같은 이름의 파일 또는 폴더가 있습니다: " + path);
            return path;
        }

        public static string Hash(string path)
        {
            using (SHA256 sha = SHA256.Create())
            using (FileStream stream = File.OpenRead(path))
                return Convert.ToBase64String(sha.ComputeHash(stream));
        }

        public static void CopyVerified(string source, string destination)
        {
            bool created = false;
            try
            {
                using (FileStream input = new FileStream(source, FileMode.Open, FileAccess.Read, FileShare.Read))
                using (FileStream output = new FileStream(destination, FileMode.CreateNew, FileAccess.Write, FileShare.None))
                {
                    created = true;
                    input.CopyTo(output);
                    output.Flush(true);
                }
                if (Hash(source) != Hash(destination)) throw new IOException("사진 복사 검증에 실패했습니다.");
            }
            catch
            {
                if (created) { try { File.Delete(destination); } catch { } }
                throw;
            }
        }
    }

        public static List<string[]> ParseAlarms(string text)
        {
            List<string[]> result = new List<string[]>();
            HashSet<string> unique = new HashSet<string>();
            foreach (string raw in (text ?? "").Split('\n'))
            {
                string line = raw.Trim(); if (line.Length == 0) continue;
                string[] parts = line.Split(new char[] { '|' }, 2);
                DateTime parsed;
                if (parts.Length != 2 || parts[1].Trim().Length == 0
                    || !DateTime.TryParseExact(parts[0].Trim(), "HH:mm", CultureInfo.InvariantCulture,
                        DateTimeStyles.None, out parsed) || !unique.Add(line))
                    throw new FormatException("일정 형식은 HH:mm|일정 이름 입니다. 같은 줄을 중복하지 마세요.");
                result.Add(new string[] { parts[0].Trim(), parts[1].Trim() });
            }
            return result;
        }

        public static void Validate(Settings settings)
        {
            if (settings == null || !settings.Configured || String.IsNullOrWhiteSpace(settings.Name)
                || String.IsNullOrWhiteSpace(settings.CsvFolder) || !Path.IsPathRooted(settings.CsvFolder)
                || String.IsNullOrWhiteSpace(settings.PhotoFolder) || !Path.IsPathRooted(settings.PhotoFolder)
                || settings.Kinds == null || settings.Kinds.Count == 0 || settings.Kinds.Count > 50)
                throw new FormatException("이름, 출력 폴더와 1~50개의 항목을 설정하세요.");
            HashSet<string> ids = new HashSet<string>();
            HashSet<string> names = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
            foreach (Kind kind in settings.Kinds)
                if (kind == null || String.IsNullOrWhiteSpace(kind.Name) || String.IsNullOrWhiteSpace(kind.Id)
                    || !ids.Add(kind.Id) || !names.Add(kind.Name) || String.IsNullOrWhiteSpace(kind.Unit))
                    throw new FormatException("유형 이름, 단위와 ID를 확인하세요. 유형 이름은 중복할 수 없습니다.");
            ParseAlarms(settings.Alarms);
        }

        void UpdatePhotoStatus(Entry entry)
        {
            int missing = 0, mismatch = 0;
            for (int i = 0; i < entry.Photos.Count; i++)
            {
                string path = entry.Photos[i];
                if (!File.Exists(path)) missing++;
                else if (!String.Equals(Path.GetFileName(path), Data.PhotoFileName(entry, entry.Photos.Count,
                    i + 1, Path.GetExtension(path)), StringComparison.OrdinalIgnoreCase)) mismatch++;
            }
            photoStatus.Text = "연결 사진 " + entry.Photos.Count + " / 없음 " + missing + " / 이름 갱신 필요 " + mismatch;
        }

        void SaveEntry(object sender, EventArgs e)
        {
            if (!EnsureDay(true)) return;
            if (date.Value > DateTime.Now) { MessageBox.Show(this, "미래 시각은 입력할 수 없습니다."); return; }
            if (String.IsNullOrWhiteSpace(title.Text) || cart.Items.Count == 0)
            { MessageBox.Show(this, "제목과 하나 이상의 항목을 입력하세요."); return; }
            Entry value = new Entry(); value.Id = editing.Length == 0 ? Guid.NewGuid().ToString("N") : editing;
            value.Time = date.Value; value.Title = title.Text.Trim(); value.Note = note.Text; value.Complete = complete.Checked;
            foreach (Item item in cart.Items) value.Items.Add(new Item {
                KindId = item.KindId, Name = item.Name, Unit = item.Unit, Count = item.Count });
            try
            {
                Book next = Data.Clone(book); Entry old = Find(next, value.Id);
                if (old != null)
                {
                    value.Photos = new List<string>(old.Photos); Data.Queue(next, old.Time); next.Entries.Remove(old);
                }
                next.Entries.Add(value); Data.Queue(next, value.Time); Persist(next);
            }
            catch (Exception ex) { status.Text = "원본 저장 실패 / 입력 유지"; Error(ex); return; }
            editing = value.Id; dirty = false; SyncCsv();
            string extra = "";
            if (value.Photos.Count > 0 && MessageBox.Show(this,
                "기록은 저장되었습니다. 연결 사진의 이름도 지금 갱신합니까?", "사진명 확인",
                MessageBoxButtons.YesNo) == DialogResult.Yes) extra = RenameCore(value.Id);
            bool copied = false;
            if (book.Settings.CopyText)
            {
                try { Clipboard.SetText(value.Time.ToString("yyyy-MM-dd HH:mm") + " " + value.Title + " " + value.Describe()); copied = true; }
                catch (Exception ex) { extra += "\r\n클립보드 복사 실패: " + ex.Message; }
            }
            RefreshRecords();
            Entry saved = Find(book, editing); if (saved != null) UpdatePhotoStatus(saved);
            string message = "원본 저장 완료" + (exportError.Length == 0 ? " / CSV 갱신 완료" : " / CSV 갱신 대기")
                + (copied ? " / 문장 복사 완료" : "") + extra;
            status.Text = message;
            if (book.Settings.ShowSuccess || extra.Length > 0 || exportError.Length > 0)
                MessageBox.Show(this, message + (exportError.Length == 0 ? "" : "\r\n" + exportError), "저장 결과");
        }

        void FilterChanged(object sender, EventArgs e) { if (records != null) RefreshRecords(); }
        void RefreshRecords()
        {
            loading = true; records.Items.Clear();
            List<Entry> ordered = new List<Entry>(book.Entries);
            ordered.Sort(delegate(Entry a, Entry b) { return b.Time.CompareTo(a.Time); });
            foreach (Entry entry in ordered)
            {
                if ((entry.Complete && !showDone.Checked) || (!entry.Complete && !showActive.Checked)) continue;
                if (!allDays.Checked && entry.Time.Date != viewDate.Value.Date) continue;
                if ((entry.Title + " " + entry.Note + " " + entry.Describe()).IndexOf(search.Text, StringComparison.OrdinalIgnoreCase) < 0) continue;
                records.Items.Add(entry); if (entry.Id == editing) records.SelectedItem = entry;
            }
            loading = false;
        }

        void AttachPhotos(object sender, EventArgs e)
        {
            Entry entry = Find(book, editing);
            if (entry == null || dirty) { MessageBox.Show(this, "사진을 연결할 기록을 먼저 저장하세요."); return; }
            using (OpenFileDialog dialog = new OpenFileDialog())
            {
                dialog.Multiselect = true; dialog.Filter = "사진|*.jpg;*.jpeg;*.png;*.bmp";
                if (dialog.ShowDialog(this) != DialogResult.OK) return;
                busy = true; int moved = 0, copied = 0, failed = 0; List<string> errors = new List<string>();
                try
                {
                    string folder = Path.Combine(book.Settings.PhotoFolder, Data.Day(entry.Time)); Directory.CreateDirectory(folder);
                    foreach (string source in dialog.FileNames)
                    {
                        string destination = ""; bool linked = false, owned = false;
                        try
                        {
                            foreach (Entry other in book.Entries)
                                foreach (string path in other.Photos)
                                    if (String.Equals(Path.GetFullPath(path), Path.GetFullPath(source), StringComparison.OrdinalIgnoreCase))
                                        throw new IOException("이미 연결된 사진입니다.");
                            int total = entry.Photos.Count + dialog.FileNames.Length;
                            int ordinal = Find(book, entry.Id).Photos.Count + 1;
                            destination = Data.PhotoDestination(entry, total, ordinal, folder,
                                Path.GetExtension(source).ToLowerInvariant(), null);
                            Data.CopyVerified(source, destination); owned = true;
                            Book next = Data.Clone(book); Find(next, entry.Id).Photos.Add(destination);
                            Data.Queue(next, entry.Time); Persist(next); linked = true;
                            // Recheck immediately before removing the source. On failure, keep both files.
                            if (Data.Hash(source) != Data.Hash(destination)) throw new IOException("원본이 변경되어 삭제하지 않았습니다.");
                            File.Delete(source); moved++;
                        }
                        catch (Exception ex)
                        {
                            if (linked) copied++; else failed++;
                            if (!linked && owned) { try { if (File.Exists(destination)) File.Delete(destination); } catch { } }
                            errors.Add(Path.GetFileName(source) + ": " + ex.Message);
                        }
                    }
                }
                catch (Exception ex) { failed = dialog.FileNames.Length - moved - copied; errors.Add(ex.Message); }
                finally { busy = false; }
                string renamed = RenameCore(entry.Id);
                UpdatePhotoStatus(Find(book, entry.Id)); SyncCsv();
                MessageBox.Show(this, "이동 " + moved + " / 연결·복사만 완료 " + copied + " / 실패 " + failed
                    + "\r\n" + String.Join("\r\n", errors.ToArray()) + renamed, "사진 처리 결과");
            }
        }
        void RenamePhotos(object sender, EventArgs e)
        {
            if (dirty) { MessageBox.Show(this, "기록을 먼저 저장하세요."); return; }
            if (Find(book, editing) == null) return;
            MessageBox.Show(this, RenameCore(editing), "사진명 갱신 결과");
            UpdatePhotoStatus(Find(book, editing));
        }
        string RenameCore(string id)
        {
            busy = true; int renamed = 0, missing = 0; List<string> errors = new List<string>();
            try
            {
                Entry entry = Find(book, id);
                for (int i = 0; i < entry.Photos.Count; i++)
                {
                    string source = Find(book, id).Photos[i];
                    if (!File.Exists(source)) { missing++; continue; }
                    string expected = Data.PhotoFileName(entry, entry.Photos.Count, i + 1, Path.GetExtension(source));
                    if (String.Equals(Path.GetFileName(source), expected, StringComparison.OrdinalIgnoreCase)) continue;
                    string destination = "";
                    bool moved = false;
                    try
                    {
                        destination = Data.PhotoDestination(entry, entry.Photos.Count, i + 1,
                            Path.GetDirectoryName(source), Path.GetExtension(source), source);
                        File.Move(source, destination); moved = true;
                        Book next = Data.Clone(book); Find(next, id).Photos[i] = destination; Persist(next); renamed++;
                    }
                    catch (Exception ex)
                    {
                        string rollback = "";
                        if (moved) { try { File.Move(destination, source); } catch (Exception failure) { rollback = " / 되돌리기 실패: " + destination + " / " + failure.Message; } }
                        errors.Add(Path.GetFileName(source) + ": " + ex.Message + rollback);
                    }
                }
            }
            catch (Exception ex) { errors.Add(ex.Message); }
            finally { busy = false; }
            return "\r\n이름 변경 " + renamed + " / 파일 없음 " + missing + " / 오류 " + errors.Count
                + (errors.Count == 0 ? "" : "\r\n" + String.Join("\r\n", errors.ToArray()));
        }

        bool EnsureDay(bool force)
        {
            DateTime today = DateTime.Now.Date;
            if (book.AcknowledgedDate == today) { editor.Enabled = true; return true; }
            if (!force && promptedDate == today) return false;
            promptedDate = today; editor.Enabled = false; busy = true;
            try
            {
                MessageBox.Show(this, "날짜가 " + Data.Day(today) + "입니다.\r\n"
                    + "기존 기록은 복제하지 않고 보존합니다. 입력 중인 내용과 일시는 그대로 유지됩니다.\r\n"
                    + "과거 기록은 날짜 필터 또는 [모든 날짜]에서 조회하세요.", "날짜 확인");
                Book next = Data.Clone(book); next.AcknowledgedDate = today;
                next.AlarmAcknowledgements.RemoveAll(delegate(string key) { return !key.StartsWith(Data.Day(today) + "|", StringComparison.Ordinal); });
                Persist(next); editor.Enabled = true; return true;
            }
            catch (Exception ex) { status.Text = "날짜 확인 저장 실패 / 입력 잠김 / 날짜 확인 버튼으로 재시도"; Error(ex); return false; }
            finally { busy = false; }
        }
        void Tick(object sender, EventArgs e)
        {
            if (busy || ContainsModal()) return;
            if (!EnsureDay(false)) return;
            try
            {
                DateTime now = DateTime.Now; List<string> due = new List<string>(); List<string> keys = new List<string>();
                foreach (string[] alarm in ConfigDialog.ParseAlarms(book.Settings.Alarms))
                {
                    string key = Data.Day(now) + "|" + alarm[0] + "|" + alarm[1];
                    DateTime at = now.Date.Add(TimeSpan.ParseExact(alarm[0], @"hh\:mm", CultureInfo.InvariantCulture));
                    if (now >= at && !book.AlarmAcknowledgements.Contains(key))
                    { due.Add(alarm[0] + " " + alarm[1]); keys.Add(key); }
                }
                if (due.Count == 0) return;
                busy = true;
                MessageBox.Show(this, "확인하지 않은 금일 일반 일정입니다.\r\n" + String.Join("\r\n", due.ToArray()), "일정 알림");
                Book next = Data.Clone(book); next.AlarmAcknowledgements.AddRange(keys); Persist(next);
            }
            catch (Exception ex) { Error(ex); }
            finally { busy = false; }
        }
